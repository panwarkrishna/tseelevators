RewriteEngine On
RewriteBase /

# mail.php
RewriteCond %{REQUEST_URI} ^/mail\.php$ [NC]
RewriteRule ^ - [L]

# Remove trailing slash
RewriteCond %{REQUEST_URI} ^(.+)/$
RewriteRule ^ %1 [R=301,L]

# Don't rewrite existing files
RewriteCond %{REQUEST_FILENAME} -f
RewriteRule ^ - [L]

# Next.js fallback
RewriteRule ^ /index.html [L]
