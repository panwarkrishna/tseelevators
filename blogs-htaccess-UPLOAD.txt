# ============================================
# public_html/blogs/.htaccess  (WordPress at /blogs)
# Overrides root Next.js settings for this folder only.
# ============================================

# Root .htaccess sets "DirectorySlash Off" - WordPress needs it ON,
# otherwise /blogs/wp-admin gives 403.
DirectorySlash On
DirectoryIndex index.php index.html
Options -Indexes

# Block direct access to sensitive files
<FilesMatch "^(wp-config\.php|readme\.html|license\.txt|\.htaccess)$">
  Require all denied
</FilesMatch>

# /blogs (no slash) -> /blogs/  (kept OUTSIDE the WordPress block so
# saving Permalinks never removes it)
<IfModule mod_rewrite.c>
RewriteEngine On
RewriteBase /blogs/
RewriteCond %{REQUEST_URI} ^/blogs$ [NC]
RewriteRule ^ https://tseelevators.com/blogs/ [L,R=301]

# Folder without slash -> add slash (/blogs/wp-admin -> /blogs/wp-admin/)
RewriteCond %{REQUEST_FILENAME} -d
RewriteRule ^(.+[^/])$ https://tseelevators.com/blogs/$1/ [L,R=301]

# Serve index.php for folder URLs directly (fixes 403 on /blogs/ and
# /blogs/wp-admin/ caused by root "DirectorySlash Off" / DirectoryIndex)
RewriteRule ^$ index.php [L]
RewriteCond %{REQUEST_FILENAME} -d
RewriteCond %{REQUEST_FILENAME}/index.php -f
RewriteRule ^(.+)/$ $1/index.php [L]
</IfModule>

# BEGIN WordPress
<IfModule mod_rewrite.c>
RewriteEngine On
RewriteBase /blogs/
RewriteRule .* - [E=HTTP_AUTHORIZATION:%{HTTP:Authorization}]
RewriteRule ^index\.php$ - [L]
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule . /blogs/index.php [L]
</IfModule>
# END WordPress
